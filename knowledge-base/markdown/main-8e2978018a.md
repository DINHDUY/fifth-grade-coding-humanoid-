import json
import Hiwonder
import Hiwonder_IIC
import time

# MicroPython 下用 ticks_ms 做计时，避免 time.time() 不递增导致无法触发 LLM 请求
def _elapsed_ms(start_ticks):
    return time.ticks_diff(time.ticks_ms(), start_ticks)

def _now_ticks():
    return time.ticks_ms()

# ---------- 硬件初始化（I2C 400000 用于 WonderLLM 视觉长串传输；不支持时回退默认）----------
try:
    iic = Hiwonder_IIC.IIC(1, 23, 22, 400000)
except Exception:
    iic = Hiwonder_IIC.IIC()
tony = Hiwonder.Tonybot()
sonar = Hiwonder_IIC.I2CSonar(iic)
cam = Hiwonder_IIC.ESP32S3Cam(iic)
sonar.setRGB(0, 0, 0, 0)

# WonderLLM 模块 I2C 地址
WONDERLLM_ADDR = 0x55

# LLM 初始化等待超时（毫秒）一致需给模块配网/启动时间
_LLM_INIT_TIMEOUT_MS = 30000
_LLM_INIT_POLL_MS = 500

# ---------- 动作组编号----------
TURN_RIGHT = 66
TURN_LEFT = 65
GO_STRAIGHT = 63
RETREAT = 22
TRANSITION = 18

# ---------- 巡线阈值----------
LEFT_MIN = 240 // 2 - 40
RIGHT_MAX = 240 // 2 + 40

# ---------- 阶段枚举----------
STAGE_LINETRACKING = 0
STAGE_WAITING = 1
STAGE_CUSTOM = 2

# 初始化机器人
tony.runActionGroup(0, 1)
tony.attachHead()
tony.moveHeadAngle(90)
time.sleep_ms(200)
sonar.setRGB(0, 0, 0, 0)
time.sleep_ms(2000)

print("start.")

# 状态机
stage = STAGE_LINETRACKING
step = 0
tickstart1 = 0
tickstart2 = 0

# 大模型MCP提示词
VISION_PROMPT = "识别前方画面,如果前方有信号灯且信号灯亮绿灯就只返回true,没有就只返回false,不要返回任何其他信息"

# WonderLLM 视觉结果缓存（轮询到的 json_data_raw 原文）
last_vision_json_raw = None
# 调试：等待阶段轮询次数，用于间隔打印
_debug_wait_poll_count = 0


def body_follow(x):
    """机体运动函数"""
    global step
    print(x)
    if not tony.isRunning():
        if step == 0:
            if LEFT_MIN < x < RIGHT_MAX:
                tony.runActionGroup(GO_STRAIGHT, 1)
                time.sleep_ms(850)
            elif 0 < x < LEFT_MIN:
                tony.runActionGroup(TURN_LEFT, 2)
                time.sleep_ms(2700)
                step = 1
            else:
                tony.runActionGroup(TURN_RIGHT, 2)
                time.sleep_ms(2850)
                step = 2
        elif step == 1:
            if 0 < x < LEFT_MIN:
                tony.runActionGroup(TURN_LEFT, 1)
                time.sleep_ms(1350)
            else:
                tony.runActionGroup(GO_STRAIGHT, 1)
                time.sleep_ms(900)
                step = 0
        elif step == 2:
            if x > RIGHT_MAX:
                tony.runActionGroup(TURN_RIGHT, 1)
                time.sleep_ms(1450)
            else:
                tony.runActionGroup(GO_STRAIGHT, 1)
                time.sleep_ms(900)
                step = 0
        else:
            step = 0


def read_black_block():
    """读取黑色块（巡线用）。返回中心 x 或 None。"""
    result = cam.read_color(1)  # 第一路颜色寄存器，对应黑色
    if result is not None and len(result) >= 4:
        x = result[0] + result[2] // 2
        return x
    return None


def read_yellow_block():
    """读取黄色块。返回 (x, y, w, h) 或 None。"""
    result = cam.read_color(3)  # 第三路颜色寄存器，对应黄色
    if result is not None and len(result) >= 4:
        return result
    return None


# ---------- WonderLLM 摄像头视觉：单次调用内连续读完全部分片）----------
# 协议：帧头 0xAA 0x55 或 0xAB 变体；data_len(2B) part_ID(2B) part_num(2B)；每包 body=data_len+1 字节（最后 1 字节 XOR 校验）
_WONDERLLM_MAX_PARTS = 64
_WONDERLLM_MAX_FRAME = 512

def _xor_checksum(data):
    c = 0
    for b in data:
        c ^= b
    return c & 0xFF


def _parse_llm_header(header):
    """解析 8 字节包头，与 Arduino receive_frame_head 一致。返回 (data_len, part_id, part_num, is_ab_fmt) 或 None。"""
    if len(header) != 8:
        return None
    if header[0] == 0xAA and header[1] == 0x55:
        data_len = (header[2] << 8) | header[3]
        part_id = (header[5] << 8) | header[4]
        part_num = (header[7] << 8) | header[6]
        return (data_len, part_id, part_num, False)
    if header[0] == 0xAB and header[1] == 0xAA and header[2] == 0x55:
        data_len = (header[3] << 8) | header[4]
        part_id = header[5] | (header[6] << 8)
        part_num = header[7]
        return (data_len, part_id, part_num, True)
    if header[0] == 0xAB and header[1] == 0x00:
        data_len = (header[2] << 8) | header[3]
        part_id = (header[4] << 8) | header[5]
        part_num = (header[6] << 8) | header[7]
        return (data_len, part_id, part_num, True)
    return None


def wonderllm_request_vision(prompt):
    """发送 vision 请求。"""
    obj = {"tool_name": "mcu.request", "command": "vision", "params": prompt}
    try:
        raw = json.dumps(obj).encode("utf-8")
        iic.writeto(WONDERLLM_ADDR, raw)
        time.sleep_ms(5)
        print("LLM vision requested")
    except Exception as e:
        print("LLM vision request err:", e)


def _wonderllm_read_one_payload():
    """
    单次调用内从第一包开始连续读完全部分片后返回。
    - 只处理 part_id==1 的新帧，否则返回 None。
    - 每包：delay_ms(2) -> 读 data_len+1 字节 -> 校验 XOR（0xAB 格式校验失败仍用数据）。
    - 多包时：delay_ms(5) -> 读下一包头 -> 校验 part_id 连续、part_num 一致 -> 继续。
    - 延时用 2ms/5ms 避免长时间阻塞触发看门狗。
    """
    try:
        header = iic.readfrom(WONDERLLM_ADDR, 8)
        parsed = _parse_llm_header(header)
        if parsed is None:
            return None
        data_len, part_id, part_num, is_ab_fmt = parsed
        if data_len <= 0 or data_len > 255:
            return None
        if part_num <= 0 or part_num > _WONDERLLM_MAX_PARTS:
            return None
        if part_id != 1:
            return None
        buffer = []
        total_len = 0
        for i in range(part_num):
            time.sleep_ms(2)
            body = iic.readfrom(WONDERLLM_ADDR, data_len + 1)
            if len(body) != data_len + 1:
                return None
            data = body[:-1]
            if not is_ab_fmt and _xor_checksum(data) != body[-1]:
                return None
            buffer.append(data)
            total_len += len(data)
            if total_len > _WONDERLLM_MAX_FRAME:
                return None
            if i + 1 < part_num:
                time.sleep_ms(5)
                head2 = iic.readfrom(WONDERLLM_ADDR, 8)
                p2 = _parse_llm_header(head2)
                if p2 is None:
                    return None
                data_len_next, part_id_next, part_num_next, _ = p2
                if part_id_next != part_id + 1 or part_num_next != part_num or data_len_next <= 0 or data_len_next > 255:
                    return None
                part_id = part_id_next
                data_len = data_len_next
        merged = b"".join(buffer)
        print("[dbg] llm_rx: ok len=%d" % len(merged))
        return merged
    except Exception as e:
        print("[dbg] llm_rx ex:", e)
        return None


_llm_rx_buffer = b""
_LLM_RX_BUFFER_MAX = 256

def wonderllm_poll_vision_result():
    """轮询 WonderLLM 视觉结果；若设备逐字节发(多段 len=1)，先拼到 buffer 再解码。"""
    global last_vision_json_raw, _llm_rx_buffer
    payload = _wonderllm_read_one_payload()
    if payload is None or len(payload) == 0:
        return None
    _llm_rx_buffer += payload
    if len(_llm_rx_buffer) > _LLM_RX_BUFFER_MAX:
        _llm_rx_buffer = _llm_rx_buffer[-_LLM_RX_BUFFER_MAX:]
    try:
        raw_str = _llm_rx_buffer.decode("utf-8")
    except Exception:
        return None
    raw_str = raw_str.strip()
    if len(raw_str) < 5 or "vision" not in raw_str:
        return None
    start = raw_str.find("{")
    if start < 0 or "}" not in raw_str:
        return None
    end = raw_str.rfind("}")
    if end < start:
        return None
    candidate = raw_str[start : end + 1]
    if "vision" not in candidate:
        return None
    try:
        obj = json.loads(candidate)
        v = obj.get("vision")
        if v not in (True, False, "true", "false"):
            return None
    except Exception:
        return None
    consumed = raw_str[: end + 1]
    _llm_rx_buffer = _llm_rx_buffer[len(consumed.encode("utf-8")) :].lstrip(b" \t\n\r")
    last_vision_json_raw = candidate
    show = candidate[:100] + "..." if len(candidate) > 100 else candidate
    print("[dbg] llm_result:", show)
    print("[dbg] has_true:", "true" in candidate)
    return candidate


def check_green_light_vision():
    """
    使用 WonderLLM 摄像头的视觉识别结果判断是否为绿灯。
    仅当 vision 明确为 "true" 或 True 时返回 True，其它（"false"、长文本、畸形值）一律返回 False。
    """
    global last_vision_json_raw
    if last_vision_json_raw is None:
        return False
    s = last_vision_json_raw.strip()
    if not s.startswith("{"):
        return False
    try:
        obj = json.loads(s)
        v = obj.get("vision")
        if v is True:
            return True
        if isinstance(v, str) and v.strip().lower() == "true":
            return True
    except Exception:
        pass
    return False


def clear_vision_result():
    """识别到绿灯并进入下一阶段后清空缓存，避免重复触发。"""
    global last_vision_json_raw
    last_vision_json_raw = None

def wait_llm_ready():
    """等待 WonderLLM 初始化完成。"""
    print("waiting for LLM init...")
    end_tick = time.ticks_add(time.ticks_ms(), _LLM_INIT_TIMEOUT_MS)
    while time.ticks_diff(end_tick, time.ticks_ms()) > 0:
        try:
            devices = iic.scan()
            if WONDERLLM_ADDR in devices:
                time.sleep_ms(2000)
                print("LLM ready.")
                return True
        except Exception:
            pass
        time.sleep_ms(_LLM_INIT_POLL_MS)
    print("LLM init timeout, delay 13s ...")
    time.sleep_ms(13000)
    print("LLM init done.")
    return False


# ---------- 主循环 ----------
wait_llm_ready()
while True:
    if stage == STAGE_LINETRACKING:
        # 巡线：黑线跟踪
        x = read_black_block()
        if x is not None:
            print("find black")
            body_follow(x)

        # 黄块检测：满足条件则进入等待阶段
        blue = read_yellow_block()
        if blue is not None and len(blue) >= 2 and blue[1] >= 120:
            print("find yellow")
            stage = STAGE_WAITING
            tickstart1 = _now_ticks()
            tickstart2 = _now_ticks()
            _llm_rx_buffer = b""
            last_vision_json_raw = None
            wonderllm_request_vision(VISION_PROMPT)
            time.sleep_ms(500)

    elif stage == STAGE_WAITING:
        # 等待阶段：用 WonderLLM 摄像头做视觉识别，识别到绿灯后通过（使用 ticks_ms 保证计时有效）
        _debug_wait_poll_count += 1
        if _elapsed_ms(tickstart1) >= 1000:
            wonderllm_request_vision(VISION_PROMPT)
            tickstart1 = _now_ticks()

        if _elapsed_ms(tickstart2) >= 50:
            wonderllm_poll_vision_result()
            if check_green_light_vision():
                print("find green light")
                clear_vision_result()
                stage = STAGE_CUSTOM
                tony.runActionGroup(1, 4)
                time.sleep_ms(4000)
            tickstart2 = _now_ticks()
            # 每约 1 秒打印一次当前状态，确认是否收到过 LLM 数据
            if _debug_wait_poll_count % 20 == 0:
                s = last_vision_json_raw

    elif stage == STAGE_CUSTOM:
        print("line follow end")
        tony.runActionGroup(0, 1)
        stage = STAGE_LINETRACKING
        print("restart line follow")

    time.sleep_ms(50)
